# What this is: the list of addresses the lifter turns into nx_hook_entry / mid-function hooks, so that
# xc2_boot.cpp's hook_add_entry calls have somewhere to fire. Without --sites the lifter emits no hook at all and
# every hook_add_entry silently does nothing.
#

# build.ps1 uses build\sites-main.txt when it exists and falls back to this file, so put your own additions
# in build\sites-main.txt and leave this one alone.
